# ==========================================
# PI ZERO 2 W SPECIFIC OVERLAYS
# ==========================================
[pi02]
# Enable USB Ethernet Gadget mode
dtoverlay=dwc2
# Underclock for heat -- Zero 2 W only, the bigger Pis keep their stock clock
arm_freq=800

# ==========================================
# PWNAGOTCHI HARDWARE BUSES (Applies to all)
# ==========================================
[all]
dtparam=spi=on
dtparam=i2c_arm=on
dtparam=sdio_overclock=25
dtoverlay=i2c-rtc,ds3231
dtoverlay=disable-bt
enable_uart=1
arm_64bit=1
kernel=kernel8-sdiofix.img

# Hardware PWM (GPIO12/13) for Display Hat Mini's backlight
dtoverlay=pwm-2chan,pin=12,func=4,pin2=13,func2=4

# ==========================================
# PI 5 SPECIFIC -- must come after [all] so it wins
# ==========================================
[pi5]
# BCM2712 has its own kernel; the sdiofix build above is the v8 one
kernel=kernel_2712.img
